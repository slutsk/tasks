#include <iostream>
#include <vector>
#include <set>
using namespace std;

int main()
{
    long long N, M, K, A, B, C;
    cin >> N >> M >> K >> A >> B >> C;
    long long mx = max({N*A, M*B, K*C});
    if (mx == N * A) cout << "Conrad ";
    if (mx == M * B) cout << "Harold ";
    if (mx == K * C) cout << "Wilhelm ";
}
