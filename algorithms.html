  1. z-функция 
    vector<int> get_z_function(const string &s)
{
    vector<int> z(s.size());
    int l = 0, r = 0;
    for (int i = 1; i < s.size(); i++)
    {
        if (i <= r)
            z[i] = min(r - i + 1, z[i - l]);
        while (z[i] + i < s.size() && s[z[i]] == s[i + z[i]])
            z[i]++;
        if (r < z[i]+i-1)
        {
            l = i;
            r = i + z[i] - 1;
        }
    }
    return z;
}
