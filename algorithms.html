  1. Минимальный делитель числа, не равный единице 
    int find_min_divisor(int n)
{
    int d = 2;
    while (d*d <= n && n % d) d++;
    return d;
}
  2. Различные делители числа 
    vector get_divisors(int n)
{
    vector divisors;
    int d = 1;
    while (d * d < n)
    {
        if (n % d == 0) {
            divisors.push_back(d);
            divisors.push_back(n / d);
        }
        d++;
    }
    if (d * d == n)
        divisors.push_back(d);
    return divisors;
}
  3. Проверка числа на простоту 
    bool is_prime(int n)
{
    if (n < 2) return false;
    for (int i = 2; i * i <= n; i++)
        if (n % i == 0)
            return false;
    return true;
}
  4. Разложение числа на простые множители (факторизация) 
    vector factorize(int n)
{
    vector fact;
    int d = 2;
    while (d * d <= n)
    {
        while (n % d == 0) {
            fact.push_back(d);
            n /= d;
        }
        d++;
    }
    if (n > 1)
        fact.push_back(n);
    return fact;
}
  5. Алгоритм Евклида (итеративный вариант) 
    vectorint gcd(int a, int b)
{
    while (a && b)
    {
        if (a > b) a %= b;
        else b %= a;
    }
    return a + b;
}

  7. Префикс-функция 
    vector get_prefix_function(const string& s)
{
    vector pref(s.size());
    for (int i = 1; i < s.size(); i++)
    {
        int j = pref[i - 1];  
        while (j > 0 && s[i] != s[j])
            j = pref[j - 1];
        if (s[i] == s[j])
            pref[i] = j + 1;
    }
    return pref;
}
  8. z-функция 
    vector<int> get_z_function(const string &s)
{
    vector<int> z(s.size());
    int l = 0, r = 0;
    for (int i = 1; i < s.size(); i++)
    {
        if (i <= r)
            z[i] = min(r - i + 1, z[i - l]);
        while (z[i] + i < s.size() && s[z[i]] == s[i + z[i]])
            z[i]++;
        if (r < z[i] + i - 1)
        {
            l = i;
            r = i + z[i] - 1;
        }
    }
    return z;
}
  9. hash-функция 
    int compute_hash(const string & str, int p = 257, int MOD = 1e9+7) {
    long long hash = 0;
    for (auto c : str) 
        hash = (hash*p + int(c) ) % MOD;
    return (int) hash;
}
  10. Префиксные хеши 
    vector compute_prefix_hashes(const string& s, int p = 257, int MOD = 1e9+7 ) {
    vector pref = { 0 };
    for (auto c : s)
        pref.push_back((1ll*pref.back()*p + c)%MOD);
    return pref;
}
